class Book:
    def __init__(self, title, author, price):
        self.title = title
        self.author = author
        self.price = price

    def display(self):
        print("Book Title :", self.title)
        print("Author :", self.author)
        print("Price :", self.price)


# Creating book objects
b1 = Book("Python Basics", "John Smith", 450)
b2 = Book("Data Structures", "Anu Kumar", 550) 

#display details
b1.display()
print("----------------------")
b2.display()
